cmake_minimum_required(VERSION 3.20)
project(volatility_susceptibility LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PYBIND11_FINDPYTHON ON)
find_package(Python3 COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(market_core core/market_core.cpp)
set_target_properties(market_core PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/python" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/python")
foreach(configuration RELEASE RELWITHDEBINFO MINSIZEREL DEBUG)
    set_target_properties(market_core PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${configuration} "${CMAKE_CURRENT_SOURCE_DIR}/python" RUNTIME_OUTPUT_DIRECTORY_${configuration} "${CMAKE_CURRENT_SOURCE_DIR}/python")
endforeach()
install(TARGETS market_core LIBRARY DESTINATION python)
enable_testing()
add_test(NAME python_tests COMMAND Python3::Interpreter -m unittest discover -s tests -v)
set_tests_properties(python_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
